# tls: byte streams, plain or TLS, on Mbed TLS over net (addons/tls/README.md).
# Mbed TLS 3.6.8 LTS from the release tarball (the git tag needs a submodule: framework/), built
# static with its default configuration, as a private dependency of braty_tls (static linking
# still passes it on, so nobody names it).
# ponytail: the default configuration builds everything; a trimmed mbedtls_config.h (TLS client,
# X.509, the common ciphers) would shrink the library.
include(FetchContent)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)    # the variables below win over Mbed TLS's options
set(BUILD_SHARED_LIBS OFF)               # normal variables: this directory and Mbed TLS only
set(USE_STATIC_MBEDTLS_LIBRARY ON)
set(USE_SHARED_MBEDTLS_LIBRARY OFF)
set(ENABLE_PROGRAMS OFF)
set(ENABLE_TESTING OFF)
set(GEN_FILES OFF)                       # the tarball has the generated sources
set(MBEDTLS_FATAL_WARNINGS OFF)          # third-party code: its warnings are not ours to fail on
if(BRATY_INSTALL)
    set(DISABLE_PACKAGE_CONFIG_AND_INSTALL OFF)  # Mbed TLS goes into the SDK next to the engine
else()
    set(DISABLE_PACKAGE_CONFIG_AND_INSTALL ON)
endif()
FetchContent_Declare(mbedtls
    URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.8/mbedtls-3.6.8.tar.bz2
    URL_HASH SHA256=bc54b818e87f49c0b3281dd85d24adaede0e63b4e822e3fbd49c005a38b9595b
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(mbedtls)
# Mbed TLS is tested as gnu C (its platform files need POSIX names that strict -std=c99 hides).
foreach(target mbedtls mbedx509 mbedcrypto everest p256m)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES C_EXTENSIONS ON)
    endif()
endforeach()

# The Mozilla CA bundle as a C array (configure time), so nothing is read from disk at run time.
set(cacert_pem ${CMAKE_CURRENT_SOURCE_DIR}/certs/cacert.pem)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${cacert_pem})
file(READ ${cacert_pem} cacert_hex HEX)
string(LENGTH "${cacert_hex}" cacert_hex_length)
math(EXPR BR_TLS_CACERT_SIZE "${cacert_hex_length} / 2")
string(REPEAT "[0-9a-f][0-9a-f]" 24 cacert_line)   # CMake regexes have no {n}
string(REGEX REPLACE "(${cacert_line})" "\\1\n    " cacert_lines "${cacert_hex}")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," BR_TLS_CACERT_BYTES "${cacert_lines}")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/br_tls_cacert.c CONTENT
"/* Generated by addons/tls/CMakeLists.txt from addons/tls/certs/cacert.pem: do not edit. */
#include \"br_tls_internal.h\"

/* The PEM text plus a NUL (Mbed TLS parses PEM only NUL-terminated). */
const unsigned char br_tls_cacert[@BR_TLS_CACERT_SIZE@ + 1] = {
    @BR_TLS_CACERT_BYTES@0x00
};
const size_t br_tls_cacert_size = @BR_TLS_CACERT_SIZE@;
" @ONLY)

braty_addon(tls SOURCES src/br_tls.c ${CMAKE_CURRENT_BINARY_DIR}/br_tls_cacert.c LINK MbedTLS::mbedtls
    PACKAGES MbedTLS)
if(BRATY_INSTALL)
    install(FILES ${mbedtls_SOURCE_DIR}/LICENSE DESTINATION licenses/MbedTLS)
    install(FILES certs/LICENSE.md DESTINATION licenses/cacert)
endif()

# The loopback TLS server of the tests (tests/tls_test_server.h): test_tls, and the http and
# websocket tests through target_link_libraries(<test> PRIVATE braty_tls_test_server).
add_library(braty_tls_test_server INTERFACE)
target_include_directories(braty_tls_test_server INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(braty_tls_test_server INTERFACE MbedTLS::mbedtls)
if(TARGET test_tls)
    target_link_libraries(test_tls PRIVATE braty_tls_test_server)
endif()
