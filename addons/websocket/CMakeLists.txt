# websocket: RFC 6455 client over tls, handshake on http's parser (addons/websocket/README.md).
# wslay 1.1.1 (vendored, third_party/wslay/, MIT) does the framing. Its lib/ sources are compiled
# into braty_websocket without this project's warning flags (third-party code). wslay's own build
# generates wslayver.h and a config.h that picks the system's socket headers for htons / ntohs /
# ntohl; ours below gives those from SDL's byte-swap macros instead, so no socket header and no
# platform #ifdef is involved.
set(wslay ${CMAKE_CURRENT_SOURCE_DIR}/third_party/wslay/lib)
set(wslay_generated ${CMAKE_CURRENT_BINARY_DIR}/wslay)
set(wslay_sources ${wslay}/wslay_event.c ${wslay}/wslay_frame.c ${wslay}/wslay_net.c ${wslay}/wslay_queue.c
    ${wslay}/wslay_stack.c)
set_source_files_properties(${wslay_sources} PROPERTIES
    COMPILE_OPTIONS "$<IF:$<C_COMPILER_ID:MSVC>,/w,-w>"
    COMPILE_DEFINITIONS HAVE_CONFIG_H)

set(PACKAGE_VERSION 1.1.1)
configure_file(${wslay}/includes/wslay/wslayver.h.in ${wslay_generated}/wslay/wslayver.h @ONLY)
if(CMAKE_C_BYTE_ORDER STREQUAL "BIG_ENDIAN")
    set(WSLAY_BYTE_ORDER "#define WORDS_BIGENDIAN 1")
else()
    set(WSLAY_BYTE_ORDER "/* little-endian: WORDS_BIGENDIAN not defined */")
endif()
file(CONFIGURE OUTPUT ${wslay_generated}/config.h CONTENT
"/* Generated by addons/websocket/CMakeLists.txt for wslay (and br_websocket_frame.c): do not edit.
 * wslay's lib/config.h.in names the system's socket headers; this one has none. */
#ifndef CONFIG_H
#define CONFIG_H

#include <SDL3/SDL_endian.h>
#include <stddef.h>
#include <sys/types.h>

/* ssize_t: POSIX has it in sys/types.h (the same type as ptrdiff_t on every target we build);
 * MSVC has none. C11 allows the repeated typedef. */
typedef ptrdiff_t ssize_t;

/* The byte order functions wslay takes from arpa/inet.h or winsock2.h. */
#undef htons
#undef ntohs
#undef ntohl
#define htons(x) SDL_Swap16BE((Uint16)(x))
#define ntohs(x) SDL_Swap16BE((Uint16)(x))
#define ntohl(x) SDL_Swap32BE((Uint32)(x))

@WSLAY_BYTE_ORDER@

#endif
" @ONLY)

braty_addon(websocket SOURCES src/br_websocket.c src/br_websocket_frame.c ${wslay_sources})
# wslay's headers and ours; http's internal header (URL parser, request writer, response parser).
set(websocket_includes ${wslay}/includes ${wslay_generated} ${CMAKE_CURRENT_SOURCE_DIR}/../http/src)
target_include_directories(braty_websocket PRIVATE ${websocket_includes} ${wslay})
if(BRATY_INSTALL)
    install(FILES ${wslay}/../COPYING DESTINATION licenses/wslay)
endif()
foreach(test test_websocket test_websocket_frame)
    if(TARGET ${test})
        target_include_directories(${test} PRIVATE ${websocket_includes})
    endif()
endforeach()

# test_websocket runs the loopback TLS server of the tls tests for its wss:// case.
if(TARGET test_websocket)
    target_link_libraries(test_websocket PRIVATE braty_tls_test_server)
endif()
